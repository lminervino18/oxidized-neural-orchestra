% Parámetros de ubicación de floats.
% Los valores por omisión de LaTeX difieren las figuras con demasiada facilidad
% y dejan páginas a medio llenar. Estos permiten empaquetar más figuras por
% página sin que el texto quede relegado.
\setcounter{topnumber}{4}
\setcounter{bottomnumber}{3}
\setcounter{totalnumber}{6}
\renewcommand{\topfraction}{0.9}
\renewcommand{\bottomfraction}{0.9}
\renewcommand{\textfraction}{0.06}
\renewcommand{\floatpagefraction}{0.85}
